# Backend Implementation Plan (2/3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `backend/`, a FastAPI service exposing listings, hourly series, seasons, forecast jobs with persisted runs, collection/scheduler control with SSE logs, and items config, replacing `app.py`.

**Architecture:** A `ListingRepository` (Postgres with CSV fallback) feeds `src.forecast` and the series endpoints; a `JobManager` runs forecast threads and collector subprocesses and fans their logs out over SSE; `RunStore` persists forecast results as JSON under `runs/`; all responses share one envelope and all inputs are Pydantic-validated.

**Tech Stack:** Python 3.13 (`/opt/anaconda3/bin/python`), FastAPI, uvicorn, sse-starlette, pydantic v2, pydantic-settings, psycopg2, pandas, pytest + httpx TestClient.

**Spec:** `docs/superpowers/specs/2026-10-09-forecast-rebuild-and-web-app-design.md` (sections 4, 7, 9, 10, 11)

## Global Constraints

- Interpreter: `/opt/anaconda3/bin/python`. Install deps once: `/opt/anaconda3/bin/python -m pip install fastapi "uvicorn[standard]" sse-starlette pydantic-settings httpx`.
- Tests: `/opt/anaconda3/bin/python -m pytest tests/backend -q` from the project root.
- Every response is `{"ok": bool, "data": ..., "error": str | null, "meta": object | null}`. Validation failures return HTTP 422 with `error` listing field problems.
- Item names must match `^[A-Za-z0-9 ]{1,64}$`; Postgres table name = `item.lower().replace(" ", "_")`.
- Timestamps in and out are ISO-8601 UTC strings.
- Branch `rebuild/forecast-and-webapp`; commit per task, no push; message format `<type>: <description>`, no attribution trailer.

## Review Focus

1. Postgres configured but unreachable at startup: the app must come up on CSV within ~3 s and `/api/status` must say `"source": "csv"`. → Task 2 test `test_make_repository_falls_back_to_csv`.
2. A season POST whose range overlaps an existing season, or whose `end` precedes `start`: 422, nothing written. → Task 3 tests `test_overlap_rejected`, `test_end_before_start_rejected`.
3. `page_size` above 500 or `page` 0 on listings: 422, not a silent clamp or a 500. → Task 5 test `test_listings_pagination_validation`.
4. Forecast POST for a season with too little data: the job ends `failed` with the `InsufficientDataError` message in `error`, and the API never returns 500. → Task 7 test `test_forecast_job_fails_cleanly_on_short_data`.
5. Two collect requests for the same item while one is running: the second gets 409 and no second process starts. → Task 8 test `test_collect_same_item_twice_409`.

```python
from __future__ import annotations

from typing import Any

from pydantic import BaseModel, Field, field_validator

ITEM_NAME_RE = r"^[A-Za-z0-9 ]{1,64}$"


def ok(data: Any, meta: dict | None = None) -> dict:
    return {"ok": True, "data": data, "error": None, "meta": meta}


def fail(error: str, meta: dict | None = None) -> dict:
    return {"ok": False, "data": None, "error": error, "meta": meta}


class SeasonIn(BaseModel):
    name: str = Field(min_length=1, max_length=64)
    start: str
    end: str | None = None
    note: str = ""


class ForecastIn(BaseModel):
    item: str = Field(pattern=ITEM_NAME_RE)
    season_id: str = Field(min_length=1)
    horizon_hours: int = Field(default=72, ge=24, le=168)
    models: list[str] = Field(default_factory=list)
    coverage: float = Field(default=0.9, ge=0.5, le=0.99)


class RunTaskIn(BaseModel):
    script_path: str = Field(min_length=1)


class ItemConfigEntry(BaseModel):
    name: str = Field(pattern=ITEM_NAME_RE)
    file: str = Field(pattern=r"^[A-Za-z0-9_]+\.py$")
    csv: str = Field(pattern=r"^[A-Za-z0-9_]+\.csv$")
    category: str = "ore"
    description: str = ""
    enabled: bool = True


class ItemsConfigIn(BaseModel):
    items: list[ItemConfigEntry]
    categories: dict[str, str] = Field(default_factory=dict)
    auto_discovery: dict[str, Any] = Field(default_factory=dict)

    @field_validator("items")
    @classmethod
    def unique_names(cls, v: list[ItemConfigEntry]) -> list[ItemConfigEntry]:
        names = [i.name for i in v]
        if len(names) != len(set(names)):
            raise ValueError("duplicate item names")
        return v
```

`backend/main.py` (first version; routers added in later tasks):
```python
from __future__ import annotations

import json
from contextlib import asynccontextmanager

from fastapi import FastAPI, HTTPException, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse

from .schemas import fail, ok
from .settings import Settings


def _format_validation(exc: RequestValidationError) -> str:
    parts = []
    for e in exc.errors():
        loc = ".".join(str(x) for x in e.get("loc", []) if x != "body")
        parts.append(f"{loc}: {e.get('msg')}")
    return "; ".join(parts) or "invalid request"


def create_app(settings: Settings | None = None, repo=None) -> FastAPI:
    settings = settings or Settings()

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        from .data.repository import make_repository
        app.state.settings = settings
        app.state.repo = repo or make_repository(settings)
        _attach_services(app)
        yield
        app.state.jobs.shutdown()

    app = FastAPI(title="Darker Market", lifespan=lifespan)

    @app.exception_handler(RequestValidationError)
    async def _validation(_: Request, exc: RequestValidationError):
        return JSONResponse(status_code=422, content=fail(_format_validation(exc)))

    @app.exception_handler(HTTPException)
    async def _http(_: Request, exc: HTTPException):
        return JSONResponse(status_code=exc.status_code, content=fail(str(exc.detail)))

    @app.exception_handler(Exception)
    async def _unhandled(_: Request, exc: Exception):
        return JSONResponse(status_code=500, content=fail(f"{type(exc).__name__}: {exc}"))

    @app.get("/api/status")
    def status(request: Request):
        st = request.app.state
        return ok({
            "source": st.repo.source_name(),
            "python": settings.python_executable,
            "items": [i.__dict__ for i in st.repo.list_items()],
            "jobs_running": [j.to_dict() for j in st.jobs.active()],
            "scheduler_running": st.scheduler.running(),
        })

    _include_routers(app)
    _mount_frontend(app, settings)
    return app


def _attach_services(app: FastAPI) -> None:
    from .data.seasons import SeasonStore
    from .services.forecast_runs import RunStore
    from .services.jobs import JobManager
    from .services.scheduler_ctl import SchedulerController
    st = app.state
    st.seasons = SeasonStore(st.settings.seasons_path)
    st.runs = RunStore(st.settings.runs_dir)
    st.jobs = JobManager()
    st.scheduler = SchedulerController(st.settings, st.jobs)


def _include_routers(app: FastAPI) -> None:
    from .api import collect, config, forecast, items, jobs, listings, scheduler, seasons, series
    for module in (items, series, listings, seasons, forecast, jobs, collect, scheduler, config):
        app.include_router(module.router, prefix="/api")


def _mount_frontend(app: FastAPI, settings: Settings) -> None:
    dist = settings.frontend_dist
    if dist is None or not (dist / "index.html").exists():
        return
    from fastapi.staticfiles import StaticFiles
    from fastapi.responses import FileResponse
    app.mount("/assets", StaticFiles(directory=dist / "assets"), name="assets")

    @app.get("/{path:path}", include_in_schema=False)
    def spa(path: str):
        candidate = dist / path
        if path and candidate.is_file():
            return FileResponse(candidate)
        return FileResponse(dist / "index.html")
```

`backend/data/repository.py`:
```python
from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Callable, Protocol

import pandas as pd

from ..settings import Settings

_ITEM_RE = re.compile(r"^[A-Za-z0-9 ]{1,64}$")


def validate_item(name: str) -> str:
    if not _ITEM_RE.match(name or ""):
        raise ValueError(f"invalid item name {name!r}: letters, digits and spaces only (1-64 chars)")
    return name


def table_name(item: str) -> str:
    return validate_item(item).lower().replace(" ", "_")


@dataclass
class ItemInfo:
    name: str
    rows: int
    first_ts: str | None
    last_ts: str | None


@dataclass(frozen=True)
class ListingQuery:
    start: pd.Timestamp | None = None
    end: pd.Timestamp | None = None
    sold: bool | None = None
    min_qty: int | None = None
    max_qty: int | None = None
    page: int = 1
    page_size: int = 100


class ListingRepository(Protocol):
    def list_items(self) -> list[ItemInfo]: ...
    def fetch(self, item: str, q: ListingQuery) -> tuple[list[dict], int]: ...
    def fetch_all(self, item: str, start: pd.Timestamp | None, end: pd.Timestamp | None) -> pd.DataFrame: ...
    def source_name(self) -> str: ...


def make_repository(settings: Settings, log: Callable[[str], None] = print) -> ListingRepository:
    from .csv_repo import CsvListingRepository
    if settings.postgres_configured:
        from .postgres_repo import PostgresListingRepository
        try:
            repo = PostgresListingRepository(settings)
            repo.probe()
            log(f"data source: postgres {settings.db_host}:{settings.db_port}/{settings.db_database}")
            return repo
        except Exception as exc:
            log(f"postgres unavailable ({exc}); falling back to CSV")
    else:
        log("postgres not configured (DB_USER empty); using CSV")
    return CsvListingRepository(settings.project_root, settings.items_config_path)
```

`backend/data/csv_repo.py`:
```python
from __future__ import annotations

import json
from pathlib import Path

import pandas as pd

from .repository import ItemInfo, ListingQuery, validate_item

COLUMNS = ["item_id", "item", "quantity", "price_per_unit", "price", "has_sold", "created_at"]


class CsvListingRepository:
    def __init__(self, csv_dir: Path, items_config_path: Path):
        self._dir = Path(csv_dir)
        self._config_path = Path(items_config_path)
        self._cache: dict[str, tuple[float, pd.DataFrame]] = {}

    def source_name(self) -> str:
        return "csv"

    def _files(self) -> dict[str, Path]:
        if not self._config_path.exists():
            return {}
        cfg = json.loads(self._config_path.read_text(encoding="utf-8"))
        return {it["name"]: self._dir / it["csv"] for it in cfg.get("items", []) if it.get("enabled", True)}

    def _frame(self, item: str) -> pd.DataFrame:
        files = self._files()
        if item not in files or not files[item].exists():
            raise KeyError(f"no CSV data for item {item!r}")
        path = files[item]
        mtime = path.stat().st_mtime
        cached = self._cache.get(item)
        if cached and cached[0] == mtime:
            return cached[1]
        df = pd.read_csv(path, usecols=lambda c: c in COLUMNS)
        df["created_at"] = pd.to_datetime(df["created_at"], utc=True, errors="coerce")
        df["has_sold"] = df["has_sold"].astype(str).str.lower().isin(["true", "1", "t"])
        df = df.dropna(subset=["created_at"]).sort_values("created_at", ascending=False).reset_index(drop=True)
        self._cache[item] = (mtime, df)
        return df

    def list_items(self) -> list[ItemInfo]:
        out = []
        for name in self._files():
            try:
                df = self._frame(name)
            except KeyError:
                out.append(ItemInfo(name, 0, None, None))
                continue
            out.append(ItemInfo(name, len(df),
                                df["created_at"].min().isoformat() if len(df) else None,
                                df["created_at"].max().isoformat() if len(df) else None))
        return out

    def _filtered(self, item: str, start, end, sold=None, min_qty=None, max_qty=None) -> pd.DataFrame:
        df = self._frame(validate_item(item))
        m = pd.Series(True, index=df.index)
        if start is not None:
            m &= df["created_at"] >= start
        if end is not None:
            m &= df["created_at"] < end
        if sold is not None:
            m &= df["has_sold"] == sold
        if min_qty is not None:
            m &= df["quantity"] >= min_qty
        if max_qty is not None:
            m &= df["quantity"] <= max_qty
        return df[m]

    def fetch(self, item: str, q: ListingQuery) -> tuple[list[dict], int]:
        df = self._filtered(item, q.start, q.end, q.sold, q.min_qty, q.max_qty)
        total = len(df)
        page = df.iloc[(q.page - 1) * q.page_size: q.page * q.page_size].copy()
        page["created_at"] = page["created_at"].dt.strftime("%Y-%m-%dT%H:%M:%S+00:00")
        page["item_id"] = page["item_id"].astype(str)
        return page.to_dict(orient="records"), total

    def fetch_all(self, item: str, start, end) -> pd.DataFrame:
        return self._filtered(item, start, end).sort_values("created_at")
```

```python
                         password=settings.db_password, port=settings.db_port,
                         connect_timeout=settings.db_connect_timeout)
        self._items_config_path = settings.items_config_path

    def source_name(self) -> str:
        return "postgres"

    def _conn(self):
        return psycopg2.connect(**self._dsn)

    def probe(self) -> None:
        with self._conn() as c, c.cursor() as cur:
            cur.execute("SELECT 1")

    def _configured_items(self) -> list[str]:
        import json
        if not self._items_config_path.exists():
            return []
        cfg = json.loads(self._items_config_path.read_text(encoding="utf-8"))
        return [it["name"] for it in cfg.get("items", []) if it.get("enabled", True)]

    def list_items(self) -> list[ItemInfo]:
        out = []
        with self._conn() as c, c.cursor() as cur:
            for name in self._configured_items():
                tbl = table_name(name)
                cur.execute("SELECT to_regclass(%s)", (tbl,))
                if cur.fetchone()[0] is None:
                    out.append(ItemInfo(name, 0, None, None))
                    continue
                cur.execute(f"SELECT COUNT(*), MIN(created_at), MAX(created_at) FROM {tbl}")
                n, lo, hi = cur.fetchone()
                out.append(ItemInfo(name, int(n),
                                    pd.Timestamp(lo, tz="UTC").isoformat() if lo else None,
                                    pd.Timestamp(hi, tz="UTC").isoformat() if hi else None))
        return out

    @staticmethod
    def _where(start, end, sold=None, min_qty=None, max_qty=None) -> tuple[str, list]:
        clauses, params = [], []
        if start is not None:
            clauses.append("created_at >= %s"); params.append(start.to_pydatetime())
        if end is not None:
            clauses.append("created_at < %s"); params.append(end.to_pydatetime())
        if sold is not None:
            clauses.append("has_sold = %s"); params.append(sold)
        if min_qty is not None:
            clauses.append("quantity >= %s"); params.append(min_qty)
        if max_qty is not None:
            clauses.append("quantity <= %s"); params.append(max_qty)
        return (" WHERE " + " AND ".join(clauses)) if clauses else "", params

    def fetch(self, item: str, q: ListingQuery) -> tuple[list[dict], int]:
        tbl = table_name(item)
        where, params = self._where(q.start, q.end, q.sold, q.min_qty, q.max_qty)
        with self._conn() as c, c.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute(f"SELECT COUNT(*) AS n FROM {tbl}{where}", params)
            total = int(cur.fetchone()["n"])
            cur.execute(f"SELECT {_COLS} FROM {tbl}{where} ORDER BY created_at DESC LIMIT %s OFFSET %s",
                        params + [q.page_size, (q.page - 1) * q.page_size])
            rows = []
            for r in cur.fetchall():
                r = dict(r)
                r["item_id"] = str(r["item_id"])
                r["created_at"] = pd.Timestamp(r["created_at"], tz="UTC").isoformat()
                rows.append(r)
        return rows, total

    def fetch_all(self, item: str, start, end) -> pd.DataFrame:
        tbl = table_name(item)
        where, params = self._where(start, end)
        with self._conn() as c:
            df = pd.read_sql(f"SELECT {_COLS} FROM {tbl}{where} ORDER BY created_at", c, params=params)
        df["created_at"] = pd.to_datetime(df["created_at"], utc=True)
        return df
```

```python
from __future__ import annotations

import json
import uuid
from pathlib import Path

import pandas as pd

from src.forecast.pipeline import Season
from ..schemas import SeasonIn


class SeasonError(ValueError):
    pass


def _ts(value: str | None) -> pd.Timestamp | None:
    if value in (None, ""):
        return None
    try:
        t = pd.Timestamp(value)
    except Exception as exc:
        raise SeasonError(f"invalid timestamp {value!r}: {exc}") from None
    return t.tz_convert("UTC") if t.tzinfo else t.tz_localize("UTC")


class SeasonStore:
    def __init__(self, path: Path):
        self._path = Path(path)

    def _read(self) -> list[dict]:
        if not self._path.exists():
            return []
        return json.loads(self._path.read_text(encoding="utf-8")).get("seasons", [])

    def _write(self, seasons: list[dict]) -> None:
        self._path.write_text(json.dumps({"seasons": seasons}, ensure_ascii=False, indent=2), encoding="utf-8")

    def list(self) -> list[dict]:
        return sorted(self._read(), key=lambda s: s["start"])

    def get(self, season_id: str) -> dict:
        for s in self._read():
            if s["id"] == season_id:
                return s
        raise KeyError(season_id)

    def _validate(self, data: SeasonIn, others: list[dict]) -> dict:
        start, end = _ts(data.start), _ts(data.end)
        if end is not None and end <= start:
            raise SeasonError("end must be after start")
        for o in others:
            o_start, o_end = _ts(o["start"]), _ts(o["end"])
            latest_start = max(start, o_start)
            earliest_end = min(end or pd.Timestamp.max.tz_localize("UTC"),
                               o_end or pd.Timestamp.max.tz_localize("UTC"))
            if latest_start < earliest_end:
                raise SeasonError(f"range overlaps season {o['name']!r}")
        return {"name": data.name, "start": start.isoformat(),
                "end": end.isoformat() if end is not None else None, "note": data.note}

    def add(self, data: SeasonIn) -> dict:
        seasons = self._read()
        record = {"id": f"s-{uuid.uuid4().hex[:8]}", **self._validate(data, seasons)}
        self._write(seasons + [record])
        return record

    def update(self, season_id: str, data: SeasonIn) -> dict:
        seasons = self._read()
        if not any(s["id"] == season_id for s in seasons):
            raise KeyError(season_id)
        others = [s for s in seasons if s["id"] != season_id]
        record = {"id": season_id, **self._validate(data, others)}
        self._write([record if s["id"] == season_id else s for s in seasons])
        return record

    def delete(self, season_id: str) -> None:
        seasons = self._read()
        if not any(s["id"] == season_id for s in seasons):
            raise KeyError(season_id)
        self._write([s for s in seasons if s["id"] != season_id])

    def to_pipeline(self) -> list[Season]:
        return [Season(s["id"], s["name"], _ts(s["start"]), _ts(s["end"])) for s in self.list()]

    def assign(self, ts: pd.Timestamp) -> str | None:
        for s in self.to_pipeline():
            if ts >= s.start and (s.end is None or ts < s.end):
                return s.id
        return None
```

```python
def resolve_range(request: Request, season_id: str | None, start: str | None, end: str | None):

        e = parse_ts(end) or e
    except Exception as exc:
```

```python
from __future__ import annotations

import asyncio
import queue
import subprocess
import threading
import uuid
from collections import deque
from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path
from typing import Callable

TERMINAL = ("succeeded", "failed")
_SENTINEL = None


def _now() -> str:
    return datetime.now(timezone.utc).isoformat()


@dataclass
class Job:
    id: str
    kind: str
    label: str
    status: str = "queued"
    created_at: str = field(default_factory=_now)
    finished_at: str | None = None
    error: str | None = None
    result_ref: str | None = None
    meta: dict = field(default_factory=dict)
    _log: deque = field(default_factory=lambda: deque(maxlen=2000), repr=False)

    def log_tail(self, n: int = 200) -> list[str]:
        return list(self._log)[-n:]

    def to_dict(self) -> dict:
        return {"id": self.id, "kind": self.kind, "label": self.label, "status": self.status,
                "created_at": self.created_at, "finished_at": self.finished_at, "error": self.error,
                "result_ref": self.result_ref, "meta": self.meta, "log_tail": self.log_tail()}


class JobManager:
    def __init__(self):
        self._jobs: dict[str, Job] = {}
        self._subs: dict[str, list[queue.Queue]] = {}
        self._procs: dict[str, subprocess.Popen] = {}
        self._lock = threading.Lock()

    # ----- registry -----
    def create(self, kind: str, label: str, meta: dict | None = None) -> Job:
        job = Job(id=uuid.uuid4().hex[:12], kind=kind, label=label, meta=meta or {})
        with self._lock:
            self._jobs[job.id] = job
        return job

    def get(self, job_id: str) -> Job:
        return self._jobs[job_id]

    def list(self) -> list[Job]:
        return sorted(self._jobs.values(), key=lambda j: j.created_at, reverse=True)

    def active(self) -> list[Job]:
        return [j for j in self.list() if j.status in ("queued", "running")]

    def find_active(self, kind: str, **meta) -> Job | None:
        for j in self.active():
            if j.kind == kind and all(j.meta.get(k) == v for k, v in meta.items()):
                return j
        return None

    # ----- events -----
    def _publish(self, job: Job, event: str, data: str) -> None:
        for q in list(self._subs.get(job.id, [])):
            q.put((event, data))

    def append_log(self, job: Job, line: str) -> None:
        job._log.append(line)
        self._publish(job, "log", line)

    def _set_status(self, job: Job, status: str, error: str | None = None, result_ref: str | None = None) -> None:
        job.status = status
        if error is not None:
            job.error = error
        if result_ref is not None:
            job.result_ref = result_ref
        if status in TERMINAL:
            job.finished_at = _now()
        self._publish(job, "status", status)
        if status in TERMINAL:
            for q in list(self._subs.get(job.id, [])):
                q.put(_SENTINEL)

    def subscribe(self, job_id: str) -> queue.Queue:
        q: queue.Queue = queue.Queue()
        with self._lock:
            self._subs.setdefault(job_id, []).append(q)
        job = self._jobs[job_id]
        for line in job.log_tail():      # replay history so late subscribers see it
            q.put(("log", line))
        q.put(("status", job.status))
        if job.status in TERMINAL:
            q.put(_SENTINEL)
        return q

    def unsubscribe(self, job_id: str, q: queue.Queue) -> None:
        with self._lock:
            if q in self._subs.get(job_id, []):
                self._subs[job_id].remove(q)

    async def events(self, job_id: str):
        q = self.subscribe(job_id)
        try:
            while True:
                try:
                    item = q.get_nowait()
                except queue.Empty:
                    await asyncio.sleep(0.2)
                    continue
                if item is _SENTINEL:
                    return
                event, data = item
                yield {"event": event, "data": data}
        finally:
            self.unsubscribe(job_id, q)

    # ----- execution -----
    def run_in_thread(self, job: Job, fn: Callable[[Job], str | None]) -> None:
        def target():
            self._set_status(job, "running")
            try:
                ref = fn(job)
                self._set_status(job, "succeeded", result_ref=ref)
            except Exception as exc:
                self.append_log(job, f"error: {type(exc).__name__}: {exc}")
                self._set_status(job, "failed", error=f"{type(exc).__name__}: {exc}")
        threading.Thread(target=target, daemon=True, name=f"job-{job.id}").start()

    def run_process(self, job: Job, argv: list[str], cwd: Path) -> None:
        def target():
            self._set_status(job, "running")
            try:
                proc = subprocess.Popen(argv, cwd=str(cwd), stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                                        text=True, encoding="utf-8", errors="replace", bufsize=1)
                self._procs[job.id] = proc
                for line in iter(proc.stdout.readline, ""):
                    self.append_log(job, line.rstrip("\n"))
                code = proc.wait()
                if code == 0:
                    self._set_status(job, "succeeded")
                else:
                    self._set_status(job, "failed", error=f"exit code {code}")
            except Exception as exc:
                self._set_status(job, "failed", error=f"{type(exc).__name__}: {exc}")
            finally:
                self._procs.pop(job.id, None)
        threading.Thread(target=target, daemon=True, name=f"job-{job.id}").start()

    def shutdown(self) -> None:
        for proc in list(self._procs.values()):
            try:
                proc.terminate()
            except Exception:
                pass
```

```python
from __future__ import annotations

import json
import math
from pathlib import Path

import pandas as pd

from src.forecast.aggregate import hourly_series

SUMMARY_KEYS = ("run_id", "created_at", "source", "beats_baseline", "skill_vs_naive")


class RunStore:
    def __init__(self, directory: Path):
        self._dir = Path(directory)
        self._dir.mkdir(parents=True, exist_ok=True)

    def _path(self, run_id: str) -> Path:
        if not run_id.replace("-", "").replace("_", "").isalnum():
            raise KeyError(run_id)
        return self._dir / f"{run_id}.json"

    def save(self, result: dict) -> str:
        self._path(result["run_id"]).write_text(json.dumps(result, ensure_ascii=False), encoding="utf-8")
        return result["run_id"]

    def get(self, run_id: str) -> dict:
        p = self._path(run_id)
        if not p.exists():
            raise KeyError(run_id)
        return json.loads(p.read_text(encoding="utf-8"))

    def delete(self, run_id: str) -> None:
        p = self._path(run_id)
        if not p.exists():
            raise KeyError(run_id)
        p.unlink()

    @staticmethod
    def summary(run: dict) -> dict:
        return {**{k: run.get(k) for k in SUMMARY_KEYS},
                "item": run["request"]["item"], "season_id": run["season"]["id"],
                "season_name": run["season"]["name"], "horizon_hours": run["request"]["horizon_hours"],
                "selected": run["selected"]["name"], "warnings": len(run.get("warnings", []))}

    def list(self) -> list[dict]:
        runs = []
        for p in self._dir.glob("*.json"):
            try:
                runs.append(self.summary(json.loads(p.read_text(encoding="utf-8"))))
            except Exception:
                continue
        return sorted(runs, key=lambda r: r["created_at"], reverse=True)

    def actuals(self, run: dict, repo) -> list[dict]:
        fc = run["forecast"]
        start = pd.Timestamp(fc[0]["ts"])
        end = pd.Timestamp(fc[-1]["ts"]) + pd.Timedelta(hours=1)
        df = repo.fetch_all(run["request"]["item"], start, end)
        if df.empty:
            return []
        series = hourly_series(df)
        wanted = {pd.Timestamp(f["ts"]) for f in fc}
        return [{"ts": ts.isoformat(), "median_price": float(r.median_price)}
                for ts, r in series.iterrows() if ts in wanted and not math.isnan(r.median_price)]
```

```python
from __future__ import annotations

import importlib.util
import subprocess
from pathlib import Path

from ..settings import Settings
from .jobs import Job, JobManager


class SchedulerController:
    def __init__(self, settings: Settings, jobs: JobManager):
        self._settings = settings
        self._jobs = jobs
        self._proc: subprocess.Popen | None = None

    def running(self) -> bool:
        return self._proc is not None and self._proc.poll() is None

    def start(self) -> bool:
        if self.running():
            return True
        script = self._settings.project_root / "src" / "scheduler" / "start_scheduler.py"
        if not script.exists():
            raise FileNotFoundError(f"missing {script}")
        self._proc = subprocess.Popen([self._settings.python_executable, str(script), "--start"],
                                      cwd=str(self._settings.project_root),
                                      stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        return True

    def stop(self) -> bool:
        if self.running():
            self._proc.terminate()
            try:
                self._proc.wait(timeout=5)
            except subprocess.TimeoutExpired:
                self._proc.kill()
        self._proc = None
        return False

    def tasks(self) -> list[dict]:
        path = self._settings.project_root / "src" / "scheduler" / "task_config.py"
        if not path.exists():
            return []
        spec = importlib.util.spec_from_file_location("task_config_dyn", path)
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)
        out = []
        for t in getattr(mod, "TASKS", []):
            sp = Path(t["script_path"])
            rel = sp if not sp.is_absolute() else sp.relative_to(self._settings.project_root) \
                if str(sp).startswith(str(self._settings.project_root)) else sp
            out.append({"name": t["name"], "script_path": str(rel), "schedule_type": t["schedule_type"],
                        "schedule_value": str(t["schedule_value"]), "enabled": t.get("enabled", True),
                        "description": t.get("description", "")})
        return out

    def resolve_script(self, script_path: str) -> Path:
        root = self._settings.project_root.resolve()
        p = (root / script_path).resolve() if not Path(script_path).is_absolute() else Path(script_path).resolve()
        if root not in p.parents:
            raise ValueError("script_path must be inside the project")
        if not p.exists():
            raise FileNotFoundError(str(p))
        return p

    def run_task(self, script_path: str) -> Job:
        p = self.resolve_script(script_path)
        job = self._jobs.create("task", f"run {p.name}", meta={"script": str(p)})
        self._jobs.run_process(job, [self._settings.python_executable, str(p)], self._settings.project_root)
        return job
```
